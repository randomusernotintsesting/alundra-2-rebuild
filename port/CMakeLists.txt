cmake_file_api(QUERY API_VERSION 1 CODEMODEL 2)

# Object libraries preserve the flat guest dispatch link semantics: every
# generated module is linked, even when reached only by its address table.
add_library(a2_core OBJECT ${A2_CORE_SOURCES})
target_sources(a2_core PRIVATE src/native_scene_light.c src/native_color_preview.c)
add_library(a2_host_sdl OBJECT ${A2_HOST_SDL_SOURCES})
add_executable(alundra2 src/main.c)
foreach(target IN ITEMS a2_core a2_host_sdl alundra2)
    target_include_directories(${target} PRIVATE src gen)
    target_compile_definitions(${target} PRIVATE
        A2_HD_FONTS=1 A2_WORLD_PRECISION=1 A2_HUD_PRESENT=1
        A2_NATIVE_ACTOR_PACKETS=1 A2_NATIVE_ACTOR_SOURCES=1
        A2_NATIVE_BILLBOARD=1 A2_DIALOGUE_PACKETS=1)
    target_link_libraries(${target} PRIVATE SDL2::SDL2 ZLIB::ZLIB)
endforeach()
target_link_libraries(alundra2 PRIVATE a2_core a2_host_sdl)
if(A2_HAVE_EXECINFO_H)
    target_compile_definitions(a2_core PRIVATE A2_HAVE_EXECINFO_H=1)
endif()

# Keep fixture-specific definitions separate from release features. Source
# properties are directory-wide, so release-only features test target identity.
set(A2_RELEASE_TARGET "$<OR:$<STREQUAL:$<TARGET_PROPERTY:NAME>,a2_core>,$<STREQUAL:$<TARGET_PROPERTY:NAME>,a2_host_sdl>,$<STREQUAL:$<TARGET_PROPERTY:NAME>,alundra2>>")
set_source_files_properties(src/gpu.c PROPERTIES COMPILE_DEFINITIONS "A2_HD_ART=1;A2_HD_PORTRAITS=1;A2_PHASE_TIMING=1")

    PROPERTIES COMPILE_DEFINITIONS A2_PHASE_TIMING=1)
set_property(SOURCE src/main.c gen/game00_005.c gen/game00_007.c
    gen/game00_008.c gen/scps_009.c gen/event00_006.c gen/event00_008.c APPEND PROPERTY COMPILE_DEFINITIONS
    "$<${A2_RELEASE_TARGET}:A2_NATIVE_ACTOR_SNAPSHOT=1>")

set_property(SOURCE src/main.c src/gpu.c APPEND PROPERTY COMPILE_DEFINITIONS
    "$<${A2_RELEASE_TARGET}:A2_INPUT_PROMPTS=1>")

    APPEND PROPERTY COMPILE_DEFINITIONS
    "$<$<AND:${A2_RELEASE_TARGET},$<BOOL:${A2_ENABLE_CINEMA_EXPANSION}>>:A2_CINEMA_EXPANSION=1>")
set_property(SOURCE src/main.c src/cdrom.c src/cd_hle.c src/paths.c
    src/game_profiles.c src/native_menu_macos.m src/menu_sdl.c src/native_voice.c

    "$<${A2_RELEASE_TARGET}:A2_NATIVE_SAVES=1>")
